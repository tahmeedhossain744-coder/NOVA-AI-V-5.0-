        async function sendMessage() {
            const input = document.getElementById('userInput');
            const text = input.value.trim();
            if (!text && !attachedFile) return;

            const userMsg = attachedFile ? `📎 [${attachedFile.name}]\n${text}` : text;
            appendMessage('user', userMsg);
            input.value = '';
            const filePayload = attachedFile ? `\n\n--- File (${attachedFile.name}) ---\n${attachedFile.content}` : '';
            removeAttachment();

            if (text.toLowerCase().startsWith('/imagine')) {
                const prompt = text.replace(/\/imagine/i, '').trim() || 'cyberpunk warrior';
                const imgUrl = `https://image.pollinations.ai/prompt/${encodeURIComponent(prompt)}?width=800&height=800&nologo=true`;
                appendRawHTML(`<p class="text-xs text-sky-400 font-mono-code mb-2">🎨 Generated: "${prompt}"</p><img src="${imgUrl}" class="rounded-lg border border-sky-500/40 max-w-xs w-full">`);
                return;
            }

            if (text.toLowerCase().startsWith('/video')) {
                const prompt = text.replace(/\/video/i, '').trim() || 'futuristic city';
                const s1 = `https://image.pollinations.ai/prompt/${encodeURIComponent(prompt + ' cinematic 8k')}?width=900&height=500&nologo=true`;
                appendMessage('ai', `🎬 **"${prompt}"** — ডানপাশের Studio-তে চালু হয়েছে!`);
                renderLiveStudioHTML(`<body style="margin:0;background:#05070B;display:flex;align-items:center;justify-content:center;height:100vh;"><img src="${s1}" style="width:92%;border:2px solid #38BDF8;border-radius:10px;"></body>`, 'AI VIDEO STUDIO');
                return;
            }

            const selectedFull = document.getElementById('modelSelector').value;
            const [provider, modelName] = selectedFull.split(':');
            const selectedLang = document.getElementById('langSelector').value;
            const sysPrompt = `You are NOVA AI ULTIMATE built for Tahmeed. ${selectedLang !== 'auto' ? `Write code in ${selectedLang.toUpperCase()}.` : 'Support all programming languages cleanly.'}`;

            try {
                let aiReply = '';

                if (provider === 'openrouter') {
                    const key = localStorage.getItem('nova_or_key');
                    if (!key) return appendMessage('ai', '⚠️ **OpenRouter API Key নেই!** বাম পাশের **Configure API Keys** বাটনে ক্লিক করে `sk-or-v1-...` Key বসাও।');
                    const res = await fetch('https://openrouter.ai/api/v1/chat/completions', {
                        method: 'POST',
                        headers: { 'Authorization': `Bearer ${key}`, 'Content-Type': 'application/json' },
                        body: JSON.stringify({
                            model: modelName,
                            messages: [{ role: 'system', content: sysPrompt }, { role: 'user', content: text + filePayload }]
                        })
                    });
                    const data = await res.json();
                    if (data.error) throw new Error(data.error.message);
                    aiReply = data.choices[0].message.content;

                } else if (provider === 'groq') {
                    const key = localStorage.getItem('nova_groq_key');
                    if (!key) return appendMessage('ai', '⚠️ **Groq API Key নেই!** বাম পাশের **Configure API Keys** বাটনে ক্লিক করে `gsk_...` Key বসাও।');
                    const res = await fetch('https://api.groq.com/openai/v1/chat/completions', {
                        method: 'POST',
                        headers: { 'Authorization': `Bearer ${key}`, 'Content-Type': 'application/json' },
                        body: JSON.stringify({
                            model: modelName,
                            messages: [{ role: 'system', content: sysPrompt }, { role: 'user', content: text + filePayload }]
                        })
                    });
                    const data = await res.json();
                    if (data.error) throw new Error(data.error.message);
                    aiReply = data.choices[0].message.content;

                } else if (provider === 'gemini') {
                    const key = localStorage.getItem('nova_gemini_key');
                    if (!key) return appendMessage('ai', '⚠️ **Gemini API Key নেই!** বাম পাশের **Configure API Keys** বাটনে ক্লিক করে `AIzaSy...` Key বসাও।');
                    const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${modelName}:generateContent?key=${key}`, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({
                            system_instruction: { parts: [{ text: sysPrompt }] },
                            contents: [{ parts: [{ text: text + filePayload }] }]
                        })
                    });
                    const data = await res.json();
                    if (data.error) throw new Error(data.error.message);
                    aiReply = data.candidates[0].content.parts[0].text;
                }

                appendMessage('ai', aiReply);
                if (ttsEnabled) window.speechSynthesis.speak(new SpeechSynthesisUtterance(aiReply.replace(/[#*`]/g, '').slice(0, 250)));

            } catch (err) {
                appendMessage('ai', `❌ **API Error:** ${err.message}`);
            }
        }

        function executeCodeBlock(lang, code) {
            togglePreviewPanel(true);
            const box = document.getElementById('previewContent');
            lang = lang.toLowerCase();

            if (lang === 'html' || lang === 'xml') {
                renderLiveStudioHTML(code, 'LIVE HTML PREVIEW');
                return;
            }
            if (lang === 'javascript' || lang === 'js') {
                let logs = [];
                try { new Function('console', code)({ log: (...a) => logs.push(a.join(' ')) }); } catch (e) { logs.push(e.message); }
                box.innerHTML = `<div class="p-4 font-mono-code text-xs"><p class="text-sky-400 mb-2">$ node script.js</p><pre class="text-emerald-400 bg-black p-3 rounded">${escapeHtml(logs.join('\n') || 'Done')}</pre></div>`;
                return;
            }
            document.getElementById('studioTitle').innerText = `${lang.toUpperCase()} TERMINAL`;
            box.innerHTML = `<div class="p-4 font-mono-code text-xs space-y-2"><p class="text-emerald-400">$ ${lang} compiler ready</p><pre class="text-sky-300 bg-black p-3 rounded border border-sky-500/30 overflow-x-auto">${escapeHtml(code)}</pre></div>`;
        }

        function appendMessage(sender, text, save = true) {
            const box = document.getElementById('chatContainer');
            const div = document.createElement('div');
            div.className = sender === 'user' ? 'flex justify-end' : 'flex gap-2.5';

            if (sender === 'user') {
                div.innerHTML = `<div class="bg-sky-500/20 border border-sky-500/40 text-white px-3.5 py-2 rounded-xl text-xs md:text-sm max-w-md whitespace-pre-wrap">${escapeHtml(text)}</div>`;
            } else {
                div.innerHTML = `
                    <div class="w-7 h-7 rounded bg-sky-500 text-slate-950 font-bold flex items-center justify-center shrink-0 text-xs mt-1">N</div>
                    <div class="glass p-3.5 rounded-xl text-xs md:text-sm prose prose-invert max-w-full overflow-x-auto">${marked.parse(text)}</div>
                `;
                div.querySelectorAll('pre').forEach(pre => {
                    const codeEl = pre.querySelector('code');
                    hljs.highlightElement(codeEl);
                    const langClass = Array.from(codeEl.classList).find(c => c.startsWith('language-')) || 'language-code';
                    const langName = langClass.replace('language-', '');

                    const toolbar = document.createElement('div');
                    toolbar.className = 'code-toolbar';
                    toolbar.innerHTML = `
                        <span class="text-sky-400 uppercase font-bold">${langName}</span>
                        <div class="flex gap-1.5">
                            <button class="tool-btn bg-emerald-500/20 text-emerald-400 border border-emerald-500/40 run-btn">▶ Run</button>
                            <button class="tool-btn bg-sky-500/20 text-sky-400 border border-sky-500/40 copy-btn-new">Copy</button>
                        </div>
                    `;
                    toolbar.querySelector('.run-btn').onclick = () => executeCodeBlock(langName, codeEl.innerText);
                    toolbar.querySelector('.copy-btn-new').onclick = (e) => {
                        navigator.clipboard.writeText(codeEl.innerText);
                        e.target.innerText = 'Copied!';
                        setTimeout(() => e.target.innerText = 'Copy', 1500);
                    };
                    pre.appendChild(toolbar);
                });
            }
            box.appendChild(div);
            box.scrollTop = box.scrollHeight;

            if (save) {
                const s = sessions.find(x => x.id === currentSessionId);
                if (s) {
                    if (s.messages.length === 0 && sender === 'user') s.title = text.slice(0, 22);
                    s.messages.push({ sender, text, isRaw: false });
                    localStorage.setItem('nova_sessions_v5', JSON.stringify(sessions));
                    renderHistory();
                }
            }
        }

        function appendRawHTML(html) {

            box.appendChild(div);
            box.scrollTop = box.scrollHeight;
        }

        function escapeHtml(str) {
            return str.replace(/[&<>"']/g, m => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[m]));
        }
